🧪 Exact Water Chemistry Targets & Sydney Tap Water AdjustmentParamètres Chimiques Exactes & Préparation de l'Eau (Calibré Sydney)

#!/usr/bin/env python3
"""
lifepulse_daemon.py — 24/7 Biosphere Telemetry, SQLite Logger & Bitaxe AxeOS Guard
Part of Regenerative Compute (regenerativecompute.org | regenerativecompute.io)

Dependencies:
    pip install pyserial requests

Usage:
    python3 lifepulse_daemon.py --port /dev/ttyACM0 --bitaxe-ip 192.168.1.142
"""

import argparse
import hashlib
import json
import sqlite3
import time
import urllib.request
from datetime import datetime, timezone

try:
    import serial
except ImportError:
    serial = None


def init_db(db_path: str = "lifepulse_telemetry.db"):
    conn = sqlite3.connect(db_path)
    conn.execute(
        """
        CREATE TABLE IF NOT EXISTS telemetry (
            id INTEGER PRIMARY KEY AUTOINCREMENT,
            ts_utc TEXT NOT NULL,
            water_c REAL NOT NULL,
            air_c REAL NOT NULL,
            rh_pct REAL NOT NULL,
            tds_ppm REAL NOT NULL,
            lux REAL NOT NULL,
            throttled INTEGER NOT NULL,
            sha256_proof TEXT NOT NULL
        )
        """
    )
    conn.commit()
    return conn


def patch_bitaxe_axeos(bitaxe_ip: str, frequency_mhz: int, core_mv: int) -> bool:
    """
    Sends an HTTP PATCH to Bitaxe AxeOS REST API (/api/system) to smoothly
    scale thermal dissipation up or down without rebooting the miner.
    """
    url = f"http://{bitaxe_ip}/api/system"
    body = json.dumps({"frequency": frequency_mhz, "coreVoltage": core_mv}).encode("utf-8")
    req = urllib.request.Request(
        url,
        data=body,
        headers={"Content-Type": "application/json"},
        method="PATCH",
    )
    try:
        with urllib.request.urlopen(req, timeout=4) as resp:
            print(f"[AXEOS] {bitaxe_ip} -> {frequency_mhz} MHz / {core_mv} mV (HTTP {resp.status})")
            return True
    except Exception as exc:
        print(f"[WARN] Could not reach Bitaxe at {bitaxe_ip}: {exc}")
        return False


def main():
    parser = argparse.ArgumentParser(description="Life-Pulse ESP32-C3 Daemon & Bitaxe Guard")
    parser.add_argument("--port", default="/dev/ttyACM0", help="USB Serial port (e.g. COM4 or /dev/ttyACM0)")
    parser.add_argument("--baud", type=int, default=115200)
    parser.add_argument("--bitaxe-ip", default="192.168.1.142", help="Local LAN IP of Bitaxe Gamma 601")
    parser.add_argument("--max-water-c", type=float, default=29.0, help="Prawn thermal safety ceiling")
    parser.add_argument("--recover-water-c", type=float, default=26.5, help="Restore full hash temp")
    args = parser.parse_args()

    if serial is None:
        raise RuntimeError("Please install pyserial first: pip install pyserial")

    conn = init_db()
    is_throttled = False
    print(f"[START] Listening on {args.port} @ {args.baud} baud | Guarding Bitaxe @ {args.bitaxe_ip}")

    with serial.Serial(args.port, args.baud, timeout=5) as ser:
        while True:
            raw_line = ser.readline().decode("utf-8", errors="ignore").strip()
            if not raw_line.startswith("{"):
                continue
            try:
                data = json.loads(raw_line)
                water_c = float(data["water_c"])
                air_c = float(data.get("air_c", 24.5))
                rh_pct = float(data.get("rh", 78.0))
                tds_ppm = float(data.get("tds", 210.0))
                lux = float(data.get("lux", 165.0))
            except (ValueError, KeyError, json.JSONDecodeError):
                continue

            ts_utc = datetime.now(timezone.utc).isoformat()
            proof_str = f"{ts_utc}|{water_c:.2f}|{air_c:.2f}|{rh_pct:.1f}|{tds_ppm:.0f}"
            sha256_hex = hashlib.sha256(proof_str.encode("utf-8")).hexdigest()

            # Closed-loop thermal hysteresis control for prawns & Wolffia
            if water_c >= args.max_water_c and not is_throttled:
                print(f"[ALERT] Water {water_c:.2f}°C >= {args.max_water_c}°C -> Throttling Bitaxe to 400 MHz!")
                if patch_bitaxe_axeos(args.bitaxe_ip, 400, 1100):
                    is_throttled = True
            elif water_c <= args.recover_water_c and is_throttled:
                print(f"[OK] Water cooled to {water_c:.2f}°C -> Restoring Bitaxe to 525 MHz!")
                if patch_bitaxe_axeos(args.bitaxe_ip, 525, 1150):
                    is_throttled = False

            conn.execute(
                """
                INSERT INTO telemetry (ts_utc, water_c, air_c, rh_pct, tds_ppm, lux, throttled, sha256_proof)
                VALUES (?, ?, ?, ?, ?, ?, ?, ?)
                """,
                (ts_utc, water_c, air_c, rh_pct, tds_ppm, lux, int(is_throttled), sha256_hex),
            )
            conn.commit()
            print(f"[{ts_utc}] H2O={water_c:.2f}C | Air={air_c:.1f}C ({rh_pct:.0f}%) | TDS={tds_ppm:.0f}ppm | SHA={sha256_hex[:12]}")
            time.sleep(0.2)


if __name__ == "__main__":
    main()
